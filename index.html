Shader effects — Metal-style on the web

import { Image, Text, ShaderLibrary, Color, Font } from 'swiftui-for-web';

const Lib = ShaderLibrary.default;

// Static — same API as SwiftUI's .colorEffect / .layerEffect / .distortionEffect
Image('photo.jpg')
  .resizable().frame({ width: 200, height: 150 })
  .colorEffect(Lib.hueRotate(90))
  .layerEffect(Lib.blur(4));

// Animated — SVG <animate> embedded in the filter graph,
// GPU-driven, no requestAnimationFrame
Image('portrait.jpg')
  .colorEffect(Lib.animatedHueRotate({ duration: 4 }))
  .layerEffect(Lib.animatedGlow({
    color: 'rgba(255,80,200,0.7)', baseRadius: 3, peakRadius: 16, duration: 2.4
  }));

Text('NEON')
  .font(Font.largeTitle).foregroundColor(Color.white)
  .layerEffect(Lib.animatedGlow({
    color: 'rgba(0,200,255,0.85)', baseRadius: 2, peakRadius: 16, duration: 1.6
  }));
